// -----------------------------------------------------------------------
// SubProceso: Fibonacci
// Retorna el n-ésimo número de la serie de Fibonacci
// -----------------------------------------------------------------------
SubProceso resultado <- Fibonacci(n)
    Definir n, resultado Como Entero

    Si n < 0 Entonces
        Escribir "Error: Fibonacci no está definido para números negativos."
        resultado <- -1
    SiNo
        Si n = 0 O n = 1 Entonces
            resultado <- n
        SiNo
            resultado <- Fibonacci(n - 1) + Fibonacci(n - 2)
        FinSi
    FinSi
FinSubProceso


// -----------------------------------------------------------------------

SubProceso resultado <- Factorial(n)
    Definir n, resultado Como Entero

    Si n < 0 Entonces
        Escribir "Error: El factorial no está definido para números negativos."
        resultado <- -1
    SiNo
        Si n = 0 O n = 1 Entonces
            resultado <- 1
        SiNo
            resultado <- Factorial(n - 1) * n
        FinSi
    FinSi
FinSubProceso


// -----------------------------------------------------------------------

SubProceso resultado <- MultiplicacionRecursiva(n, m)
    Definir n, m, resultado Como Entero

    Si m < 0 Entonces

        resultado <- -1
    SiNo

            resultado <- 0
        SiNo
            resultado <- MultiplicacionRecursiva(n, m - 1) + n
        FinSi
    FinSi
FinSubProceso


// -----------------------------------------------------------------------
// SubProceso: DivisionEnteraRecursiva
// Calcula dividendo / divisor mediante restas recursivas sucesivas
// -----------------------------------------------------------------------
SubProceso resultado <- DivisionEnteraRecursiva(dividendo, divisor)
    Definir dividendo, divisor, resultado Como Entero

    Si divisor = 0 Entonces
        Escribir "Error: No se puede dividir entre cero."
        resultado <- -1
    SiNo
        Si dividendo < 0 O divisor < 0 Entonces
            Escribir "Error: Esta implementación es solo para números positivos."
            resultado <- -1
        SiNo
            Si dividendo - divisor < 0 Entonces
                resultado <- 0
            SiNo
                resultado <- DivisionEnteraRecursiva(dividendo - divisor, divisor) + 1
            FinSi
        FinSi
    FinSi
FinSubProceso

// Calcula base ^ exponente de forma recursiva
// -----------------------------------------------------------------------
SubProceso resultado <- PotenciaRecursiva(base, exponente)
    Definir exponente Como Entero
    Definir base, resultado Como Real

    Si exponente < 0 Entonces
        Escribir "Error: No se admiten exponentes negativos en esta versión."
        resultado <- -1
    SiNo
        Si exponente = 0 Entonces
            resultado <- 1
        SiNo
            resultado <- PotenciaRecursiva(base, exponente - 1) * base
        FinSi
    FinSi
FinSubProceso


// -----------------------------------------------------------------------
// SubProceso: SerieCollatz
// Imprime la serie de Collatz a partir de n hasta llegar a 1
// -----------------------------------------------------------------------
SubProceso resultado <- SerieCollatz(n)
    Definir n, resultado Como Entero

    Si n <= 0 Entonces
        Escribir "Error: La serie de Collatz solo aplica para enteros positivos mayores a 0."
        resultado <- -1
    SiNo
        Si n = 1 Entonces
            Escribir "END!"
            resultado <- 0
        SiNo
            Si n MOD 2 = 0 Entonces
                Escribir n / 2
                resultado <- SerieCollatz(n / 2)
            SiNo
                Escribir 3 * n + 1
                resultado <- SerieCollatz(3 * n + 1)
            FinSi
        FinSi
    FinSi
FinSubProceso
